async function evidence(info: TestInfo, name: string, body: Buffer, contentType: string) {
  const directory = resolve('test-results/floor-refit-qa');
  await mkdir(directory, { recursive: true });
  const path = resolve(directory, name);
  await writeFile(path, body);
  await info.attach(name, { path, contentType });
}

async function readProject(page: Page): Promise<ProjectDocument> {
  return page.evaluate(
    (id) =>
      new Promise<ProjectDocument>((resolve, reject) => {
        const request = indexedDB.open('gongganmiri-v1');
        request.onerror = () => reject(request.error);
        request.onsuccess = () => {
          const db = request.result;
          const get = db.transaction('projects').objectStore('projects').get(id);
          get.onerror = () => {
            db.close();
            reject(get.error);
          };
          get.onsuccess = () => {
            db.close();
            resolve(get.result);
          };
        };
      }),
    page.url().split('/').at(-1)!,
  );
}

async function patches(page: Page, points: Landmark[], originalAssetId?: string): Promise<number[][][]> {
  return page.evaluate(
    async ({ points, originalAssetId }) => {
      const output = document.createElement('canvas');
      const ctx = output.getContext('2d')!;
      if (originalAssetId) {
        const blob = await new Promise<Blob>((resolve, reject) => {
          const request = indexedDB.open('gongganmiri-v1');
          request.onerror = () => reject(request.error);
          request.onsuccess = () => {
            const db = request.result;
            const get = db.transaction('assets').objectStore('assets').get(originalAssetId);
            get.onerror = () => {
              db.close();
              reject(get.error);
            };
            get.onsuccess = () => {
              db.close();
              resolve((get.result as AssetRecord).blob);
            };
          };
        });
        const bitmap = await createImageBitmap(blob);
        output.width = bitmap.width;
        output.height = bitmap.height;
        ctx.drawImage(bitmap, 0, 0);
        bitmap.close();
      } else {
        const rendered = document.querySelector<HTMLCanvasElement>('[data-testid="canvas-frame"] canvas')!;
        output.width = rendered.width;
        output.height = rendered.height;
        ctx.drawImage(rendered, 0, 0);
      }
      return points.map((point) => {
        const x = Math.min(output.width - 2, Math.max(1, Math.floor(point.x * output.width)));
        const y = Math.min(output.height - 2, Math.max(1, Math.floor(point.y * output.height)));
        const data = ctx.getImageData(x - 1, y - 1, 3, 3).data;
        return Array.from({ length: 9 }, (_, i) => [...data.slice(i * 4, i * 4 + 4)]);
      });
    },
    { points, originalAssetId },
  );
}

function blueCoverage(patch: number[][]) {
  return patch.filter(([r, g, b]) => b - r > 55 && b - g > 30).length / patch.length;
}
function maxDifference(a: number[][], b: number[][]) {
  return Math.max(...a.flatMap((pixel, i) => pixel.slice(0, 3).map((value, j) => Math.abs(value - b[i][j]))));
}

async function refit(page: Page, beforeRevision: number) {
  await page.getByRole('button', { name: '바닥 경계 다시 맞추기', exact: true }).click();
  await expect
    .poll(async () => (await readProject(page)).editRevision, { timeout: 120000 })
    .toBeGreaterThan(beforeRevision);
  await expect(page.getByTestId('auto-detection-status')).toHaveCount(0);
  await expect(page.getByTestId('save-status')).toHaveText('이 브라우저에 저장됨');
  await expect(page.locator('.editor-error')).toHaveCount(0);
  return readProject(page);
}

test('바닥 가장자리와 제품 발밑을 다시 맞추며 보호 선택권·설정·실행 취소를 보존한다', async ({
  page,
}, info) => {
  const errors: string[] = [];
  page.on('pageerror', (error) => errors.push(error.message));
  const before = await legacyProject(page);
  const beforeFloor = before.scene.surfaces.find((surface) => surface.kind === 'floor')!;
  const original = await patches(page, originalObjects, before.scene.originalAssetId);
  const legacyEdges = await patches(page, edgeFloor);
  expect(legacyEdges.filter((patch) => blueCoverage(patch) < 0.2).length).toBeGreaterThanOrEqual(4);
  await expect(page.getByLabel('보호 경계도 함께 다듬기', { exact: true })).not.toBeChecked();
  await evidence(info, 'legacy-floor-before.png', await page.screenshot(), 'image/png');

  const retainedProtection = await refit(page, before.editRevision);
  expect(retainedProtection.scene.protection).toEqual(before.scene.protection);
  expect(retainedProtection.scene.surfaces.find((surface) => surface.kind === 'floor')?.mask).not.toEqual(
    beforeFloor.mask,
  );
  const retainedContacts = await patches(page, contactFloor);
  expect(retainedContacts.every((patch) => blueCoverage(patch) < 0.2)).toBe(true);
  await page.getByRole('button', { name: '실행 취소', exact: true }).click();
  await expect.poll(async () => (await readProject(page)).scene).toEqual(before.scene);

  await page.getByLabel('보호 경계도 함께 다듬기', { exact: true }).check();
  const beforeRepair = await readProject(page);
  const repaired = await refit(page, beforeRepair.editRevision);
  const repairedFloor = repaired.scene.surfaces.find((surface) => surface.kind === 'floor')!;
  expect(repairedFloor.id).toBe(beforeFloor.id);
  expect(repairedFloor.materialVersionId).toBe(beforeFloor.materialVersionId);
  expect(repairedFloor.tile).toEqual(beforeFloor.tile);
  expect(repairedFloor.color).toEqual(beforeFloor.color);
  expect([repairedFloor.widthMm, repairedFloor.heightMm]).toEqual([
    beforeFloor.widthMm,
    beforeFloor.heightMm,
  ]);
  expect(repairedFloor.calibrated).toBe(false);
  expect(repaired.scene.surfaces.filter((surface) => surface.kind === 'wall')).toEqual(
    before.scene.surfaces.filter((surface) => surface.kind === 'wall'),
  );
  expect(repaired.scene.fixtures).toEqual(before.scene.fixtures);
  expect(repaired.scene.backgroundAssetId).toBe(before.scene.backgroundAssetId);
  expect(repaired.scene.protection).not.toEqual(before.scene.protection);
  expect(repaired.editRevision).toBe(beforeRepair.editRevision + 1);
  expect(repaired.history.past.at(-1)).toEqual(before.scene);

  const edges = await patches(page, edgeFloor),
    contacts = await patches(page, contactFloor),
    objects = await patches(page, originalObjects);
  const coverage = [...edgeFloor, ...contactFloor].map((point, i) => ({
    ...point,
    blueCoverage: blueCoverage([...edges, ...contacts][i]),
  }));
  const preserved = originalObjects.map((point, i) => ({
    ...point,
    maxDifference: maxDifference(objects[i], original[i]),
  }));
  await evidence(
    info,
    'floor-refit-report.json',
    Buffer.from(
      JSON.stringify(
        {
          coverage,
          preserved,
          beforeMask: beforeFloor.mask,
          afterMask: repairedFloor.mask,
          retainedMaterial: repairedFloor.materialVersionId,
          errors,
        },
        null,
        2,
      ),
    ),
    'application/json',
  );
  await evidence(info, 'floor-refit-after.png', await page.screenshot(), 'image/png');
  for (const point of coverage) expect(point.blueCoverage, point.name).toBeGreaterThanOrEqual(8 / 9);
  for (const point of preserved) expect(point.maxDifference, point.name).toBeLessThanOrEqual(3);
  await page.getByRole('button', { name: '실행 취소', exact: true }).click();
  await expect.poll(async () => (await readProject(page)).scene).toEqual(before.scene);
  await page.getByRole('button', { name: '다시 실행', exact: true }).click();
  await expect.poll(async () => (await readProject(page)).scene).toEqual(repaired.scene);
  await page.reload();
  await expect(page.getByTestId('save-status')).toHaveText('이 브라우저에 저장됨');
  expect((await readProject(page)).scene).toEqual(repaired.scene);
  expect(errors).toEqual([]);
});

test('바닥 경계 다시 찾기에 실패하면 기존 수동 경계와 보호 영역을 그대로 둔다', async ({ page }, info) => {
  const errors: string[] = [];
  page.on('pageerror', (error) => errors.push(error.message));
  const before = await legacyProject(page);
  const blocked: string[] = [];
  await page.context().route(/\/models\/deeplab-ade20k\/model\.json(?:\?|$)/, async (route) => {
    blocked.push(route.request().url());
    await route.abort('failed');
  });
  await page.getByLabel('보호 경계도 함께 다듬기', { exact: true }).check();
  await page.getByRole('button', { name: '바닥 경계 다시 맞추기', exact: true }).click();
  await expect(page.locator('.editor-error')).toBeVisible({ timeout: 120000 });
  await expect(page.getByTestId('auto-detection-status')).toHaveCount(0);
  await expect(page.getByRole('button', { name: '바닥 경계 다시 맞추기', exact: true })).toBeEnabled();
  expect(blocked.length).toBeGreaterThan(0);
  expect(await readProject(page)).toEqual(before);
  expect(errors).toEqual([]);
  await evidence(
    info,
    'floor-refit-failure.json',
    Buffer.from(
      JSON.stringify(
        {
          blocked,
          errorMessage: await page.locator('.editor-error').innerText(),
          originalDocumentPreserved: true,
          errors,
        },
        null,
        2,
      ),
    ),
    'application/json',
  );
});
