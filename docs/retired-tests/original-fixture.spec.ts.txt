async function saved(page: Page): Promise<ProjectDocument> {
  await expect(page.getByTestId('save-status')).toHaveText('이 브라우저에 저장됨');
  return page.evaluate(
    (id) =>
      new Promise<ProjectDocument>((resolve, reject) => {
        const request = indexedDB.open('gongganmiri-v1');
        request.onerror = () => reject(request.error);
        request.onsuccess = () => {
          const db = request.result;
          const get = db.transaction('projects').objectStore('projects').get(id);
          get.onerror = () => {
            db.close();
            reject(get.error);
          };
          get.onsuccess = () => {
            db.close();
            resolve(get.result);
          };
        };
      }),
    page.url().split('/').at(-1)!,
  );
}
async function clickAt(page: Page, canvas: Locator, x: number, y: number) {
  await canvas.scrollIntoViewIfNeeded();
  const box = await canvas.boundingBox();
  expect(box).not.toBeNull();
  await page.mouse.click(box!.x + box!.width * x, box!.y + box!.height * y);
}
async function setRange(input: Locator, value: string) {
  await input.evaluate((element, next) => {
    const range = element as HTMLInputElement;
    Object.getOwnPropertyDescriptor(HTMLInputElement.prototype, 'value')!.set!.call(range, next);
    range.dispatchEvent(new Event('input', { bubbles: true }));
    range.dispatchEvent(new PointerEvent('pointerup', { bubbles: true }));
    range.blur();
  }, value);
}
async function pixels(page: Page) {
  return page.locator('[data-testid="canvas-frame"] canvas').evaluate((element, points) => {
    const source = element as HTMLCanvasElement,
      copy = document.createElement('canvas');
    copy.width = source.width;
    copy.height = source.height;
    const ctx = copy.getContext('2d')!;
    ctx.drawImage(source, 0, 0);
    return points.map(({ x, y }) =>
      [...ctx.getImageData(Math.floor(x * copy.width), Math.floor(y * copy.height), 1, 1).data].slice(0, 3),
    );
  }, landmarks);
}
async function expectPixels(page: Page, expected: number[][]) {
  await expect
    .poll(
      async () => {
        const actual = await pixels(page);
        return actual.every((pixel, index) =>
          pixel.every((value, channel) => Math.abs(value - expected[index][channel]) <= 4),
        );
      },
      { message: `Canvas pixels should match ${JSON.stringify(expected)}` },
    )
    .toBe(true);
}
async function drawSurface(page: Page, kind: 'floor' | 'wall', top: number, bottom: number) {
  await page.getByRole('button', { name: '면 만들기 (4점)', exact: true }).click();
  await page.getByLabel('새 면 종류').selectOption(kind);
  for (const [x, y] of [
    [0.1, top],
    [0.9, top],
    [0.9, bottom],
    [0.1, bottom],
  ])
    await clickAt(page, page.getByTestId('editor-canvas'), x, y);
  await expect(page.getByRole('heading', { name: '면 속성', exact: true })).toBeVisible();
}
async function openOutline(page: Page) {
  await page.getByRole('button', { name: '기존 제품 제거 / 배경 복원', exact: true }).click();
  const dialog = page.getByRole('dialog', { name: '기존 제품 제거 및 배경 복원', exact: true });
  const canvas = dialog.getByLabel('배경 복원 캔버스', { exact: true });
  await expect.poll(() => canvas.evaluate((element) => (element as HTMLCanvasElement).width)).toBe(800);
  await dialog.getByRole('button', { name: '제품 윤곽 그리기', exact: true }).click();
  for (const [x, y] of outline) await clickAt(page, canvas, x, y);
  await dialog.getByRole('button', { name: '제품 윤곽 완료', exact: true }).click();
  return { dialog, canvas };
}

test('분리 도기 버전을 읽는 중 복원 창을 취소하면 늦은 응답이 프로젝트를 적용하지 않는다', async ({
  page,
}, info) => {
  const errors: string[] = [];
  page.on('pageerror', (error) => errors.push(error.message));
  const photo = await sharp(
    Buffer.from(
      '<svg xmlns="http://www.w3.org/2000/svg" width="800" height="600"><rect width="800" height="600" fill="#808080"/><rect x="320" y="180" width="160" height="300" fill="#ef3030"/></svg>',
    ),
  )
    .png()
    .toBuffer();
  const background = await sharp({ create: { width: 800, height: 600, channels: 3, background: '#808080' } })
    .png()
    .toBuffer();
  await page.goto('/');
  await seedTestTiles(page);
  await expect(page.getByRole('button', { name: '새 프로젝트', exact: true })).toBeEnabled();
  await page
    .getByTestId('project-upload')
    .setInputFiles({ name: '취소 경합 원본.png', mimeType: 'image/png', buffer: photo });
  await expect(page.getByTestId('editor-canvas')).toBeVisible();
  await expect(page.locator('.canvas-loading')).toHaveCount(0);
  await drawSurface(page, 'floor', 0.55, 0.9);
  await page.locator('button.material-tile').filter({ hasText: '라이트 스톤' }).click();
  const before = await saved(page),
    beforePixels = await pixels(page);
  const restore = await openOutline(page);
  await restore.dialog
    .locator('input[type="file"]')
    .setInputFiles({ name: '복원 배경.png', mimeType: 'image/png', buffer: background });
  await expect(restore.dialog.getByLabel('배경 크기', { exact: true })).toBeVisible();
  await restore.dialog
    .getByRole('checkbox', { name: '기존 제품을 이동 가능한 도기로 분리', exact: true })
    .check();
  const productName = '취소 경합 검증 도기';
  await restore.dialog.getByLabel('분리할 제품 이름', { exact: true }).fill(productName);
  await page.evaluate((productName) => {
    const state = window as unknown as {
      restoreVersionPaused?: string;
      restoreVersionReleased?: boolean;
      releaseRestoreVersion?: () => void;
    };
    const original = IDBObjectStore.prototype.get;
    IDBObjectStore.prototype.get = function (key) {
      const request = original.call(this, key);
      if (this.name !== 'versions') return request;
      // Observe the real stored version, pausing only the final metadata read after registration.
      request.addEventListener(
        'success',
        (event) => {
          const value = request.result as { id?: string; name?: string; category?: string } | undefined;
          if (value?.name !== productName || value.category === 'tile') return;
          IDBObjectStore.prototype.get = original;
          event.stopImmediatePropagation();
          state.restoreVersionPaused = value.id;
          state.releaseRestoreVersion = () => {
            state.restoreVersionReleased = true;
            request.dispatchEvent(new Event('success'));
          };
        },
        { once: true },
      );
      return request;
    };
  }, productName);
  try {
    await restore.dialog.getByRole('button', { name: '확인한 배경 적용', exact: true }).click();
    await expect
      .poll(() =>
        page.evaluate(() => (window as unknown as { restoreVersionPaused?: string }).restoreVersionPaused),
      )
      .toBeTruthy();
    await expect(restore.dialog).toBeVisible();
    await expect(restore.dialog.getByRole('button', { name: '처리 중…', exact: true })).toBeDisabled();
    const pausedVersion = await page.evaluate(
      () => (window as unknown as { restoreVersionPaused: string }).restoreVersionPaused,
    );
    expect(await saved(page)).toEqual(before);
    await restore.dialog.getByRole('button', { name: '취소', exact: true }).click();
    await expect(restore.dialog).toHaveCount(0);
    expect(await saved(page)).toEqual(before);
    await page.evaluate(async () => {
      const state = window as unknown as { releaseRestoreVersion: () => void };
      state.releaseRestoreVersion();
      await new Promise<void>((resolve) => requestAnimationFrame(() => resolve()));
    });
    expect(
      await page.evaluate(
        () => (window as unknown as { restoreVersionReleased: boolean }).restoreVersionReleased,
      ),
    ).toBe(true);
    expect(await saved(page)).toEqual(before);
    expect(await pixels(page)).toEqual(beforePixels);
    await expect(page.locator('.editor-error')).toHaveCount(0);
    await page.reload();
    await expect(page.getByTestId('editor-canvas')).toBeVisible();
    await expect(page.locator('.canvas-loading')).toHaveCount(0);
    expect(await saved(page)).toEqual(before);
    expect(await pixels(page)).toEqual(beforePixels);
    expect(errors).toEqual([]);
    const report = {
      pausedVersion,
      cancelledDuringRealVersionRead: true,
      realLateResponseReleased: true,
      sceneHistoryRevisionsAndPixelsUnchanged: true,
      reloadPreservedOriginalProject: true,
      pageErrors: errors,
    };
    await mkdir(directory, { recursive: true });
    const path = `${directory}/cancel-during-apply.json`;
    await writeFile(path, JSON.stringify(report, null, 2));
    await info.attach('cancel-during-apply.json', { path, contentType: 'application/json' });
  } finally {
    await page
      .evaluate(() => {
        const state = window as unknown as {
          releaseRestoreVersion?: () => void;
          restoreVersionReleased?: boolean;
        };
        if (!state.restoreVersionReleased) state.releaseRestoreVersion?.();
      })
      .catch(() => {});
  }
});
