import { chromium, expect, test } from '@playwright/test';
import { build } from 'esbuild';
import { createHash } from 'node:crypto';
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import path from 'node:path';
import sharp from 'sharp';

/**
 * Replays SAVED, real TripoSR meshes through the production PNG renderer.
 * No AI inference is performed here. Uniform white is a diagnostic control,
 * never a replacement presented as an AI result. Model files are not fetched.
 * SJN_MULTIVIEW_QUALITY=1 npx playwright test e2e/multiview-quality.spec.ts
 */
function readMesh(directory: string) {
  const floats = (name: string) => {
    const bytes = readFileSync(path.join(directory, `mesh-${name}.bin`));
    return new Float32Array(bytes.buffer.slice(bytes.byteOffset, bytes.byteOffset + bytes.byteLength));
  };
  const bytes = readFileSync(path.join(directory, 'mesh-indices.bin'));
  return {
    positions: floats('positions'),
    colors: floats('colors'),
    indices: new Uint32Array(bytes.buffer.slice(bytes.byteOffset, bytes.byteOffset + bytes.byteLength)),
  };
}

function inspectMesh(mesh: ReturnType<typeof readMesh>) {
  const vertices = mesh.positions.length / 3;
  const bounds = [
    [Infinity, Infinity, Infinity],
    [-Infinity, -Infinity, -Infinity],
  ];
  let nonFinite = 0,
    outOfRangeColors = 0,
    darkBlueVertices = 0;
  for (let i = 0; i < mesh.positions.length; i++) {
    const value = mesh.positions[i];
    if (!Number.isFinite(value)) nonFinite++;
    bounds[0][i % 3] = Math.min(bounds[0][i % 3], value);
    bounds[1][i % 3] = Math.max(bounds[1][i % 3], value);
    if (!Number.isFinite(mesh.colors[i]) || mesh.colors[i] < 0 || mesh.colors[i] > 1) outOfRangeColors++;
  }
  for (let i = 0; i < mesh.colors.length; i += 3) {
    const [r, g, b] = mesh.colors.subarray(i, i + 3);
    if (r < 0.55 && b > r + 0.05 && b > g + 0.03) darkBlueVertices++;
  }
  const edges = new Map<string, number>();
  const triangles = new Set<string>();
  let duplicateTriangles = 0,
    invalidIndices = 0,
    degenerateTriangles = 0,
    longestEdge = 0;
  let largeColorDeltaEdges = 0;
  for (let i = 0; i < mesh.indices.length; i += 3) {
    const face = Array.from(mesh.indices.subarray(i, i + 3));
    if (face.some((index) => index >= vertices)) {
      invalidIndices++;
      continue;
    }
    const key = face.toSorted((a, b) => a - b).join(',');
    if (triangles.has(key)) duplicateTriangles++;
    triangles.add(key);
    const a = face[0] * 3,
      b = face[1] * 3,
      c = face[2] * 3;
    const u = [0, 1, 2].map((axis) => mesh.positions[b + axis] - mesh.positions[a + axis]);
    const v = [0, 1, 2].map((axis) => mesh.positions[c + axis] - mesh.positions[a + axis]);
    const area2 = Math.hypot(u[1] * v[2] - u[2] * v[1], u[2] * v[0] - u[0] * v[2], u[0] * v[1] - u[1] * v[0]);
    if (area2 < 1e-12) degenerateTriangles++;
    for (const [first, second] of [
      [face[0], face[1]],
      [face[1], face[2]],
      [face[2], face[0]],
    ]) {
      const edgeKey = first < second ? `${first},${second}` : `${second},${first}`;
      const count = edges.get(edgeKey) ?? 0;
      edges.set(edgeKey, count + 1);
      if (count) continue;
      const delta = [0, 1, 2].map(
        (axis) => mesh.positions[first * 3 + axis] - mesh.positions[second * 3 + axis],
      );
      longestEdge = Math.max(longestEdge, Math.hypot(...delta));
      const colorDelta = [0, 1, 2].map((axis) =>
        Math.abs(mesh.colors[first * 3 + axis] - mesh.colors[second * 3 + axis]),
      );
      if (Math.max(...colorDelta) > 0.25) largeColorDeltaEdges++;
    }
  }
  const radius = Math.hypot(...bounds[0].map((value, axis) => bounds[1][axis] - value)) / 2;
  return {
    vertices,
    triangles: mesh.indices.length / 3,
    bounds,
    radius,
    nonFinite,
    outOfRangeColors,
    invalidIndices,
    duplicateTriangles,
    degenerateTriangles,
    boundaryEdges: [...edges.values()].filter((count) => count === 1).length,
    nonManifoldEdges: [...edges.values()].filter((count) => count > 2).length,
    longestEdge,
    darkBlueVertices,
    largeColorDeltaEdges,
    conservativeCameraDepth: {
      near: 0.01,
      objectMin: radius * 3,
      objectMax: radius * 5,
      far: radius * 10 + 1,
    },
  };
}

test('real saved meshes: learned colors versus uniform-white topology control', async ({}, info) => {
  test.skip(process.env.SJN_MULTIVIEW_QUALITY !== '1', 'Opt-in saved real mesh quality replay.');
  test.setTimeout(120_000);
  const fixtures = process.env.SJN_MULTIVIEW_QUALITY_MESH
    ? [
        {
          name: 'custom',
          directory: process.env.SJN_MULTIVIEW_QUALITY_MESH,
          yaw: Number(process.env.SJN_MULTIVIEW_QUALITY_YAW ?? '0'),
        },
      ]
    : [
        { name: 'white-toilet', directory: 'test-results/multiview-live-6a', yaw: 0 },
        { name: 'garden-chair', directory: 'test-results/multiview-chair-left', yaw: -45 },
      ];
  for (const fixture of fixtures)
    expect(existsSync(path.join(fixture.directory, 'mesh-positions.bin'))).toBe(true);
  const output = path.resolve(process.env.SJN_MULTIVIEW_QUALITY_OUTPUT ?? 'test-results/multiview-quality');
  mkdirSync(output, { recursive: true });
  const bundle = await build({
    stdin: {
      contents: `
        export {renderThreeViews} from './src/lib/multiview/render';
        import * as THREE from 'three';
        import {viewCameraPosition} from './src/lib/multiview/render';
        // Diagnostic normal controls, explicitly separate from the production color renderer.
        export async function renderNormalControls(mesh, yaw) {
          const geometry = new THREE.BufferGeometry();
          geometry.setAttribute('position', new THREE.BufferAttribute(mesh.positions, 3));
          geometry.setIndex(new THREE.BufferAttribute(mesh.indices, 1));
          geometry.computeVertexNormals();
          geometry.computeBoundingBox();
          const center = geometry.boundingBox.getCenter(new THREE.Vector3());
          const radius = geometry.boundingBox.getSize(new THREE.Vector3()).length() / 2;
          const frame = radius * 1.1;
          const camera = new THREE.OrthographicCamera(-frame, frame, frame, -frame, .01, radius * 10 + 1);
          camera.up.set(0, 0, 1);
          camera.position.copy(viewCameraPosition(yaw, 15, radius * 4)).add(center);
          camera.lookAt(center); camera.updateMatrixWorld();
          const renderer = new THREE.WebGLRenderer({alpha:true, antialias:true, preserveDrawingBuffer:true});
          renderer.setPixelRatio(1); renderer.setSize(768,768,false); renderer.setClearColor(0,0);
          renderer.outputColorSpace = THREE.SRGBColorSpace; renderer.toneMapping = THREE.NoToneMapping;
          const scene = new THREE.Scene(), controls = [];
          try {
            for (const flat of [false, true]) {
              const material = new THREE.MeshNormalMaterial({side:THREE.DoubleSide, flatShading:flat});
              const object = new THREE.Mesh(geometry, material); scene.add(object);
              try {
                renderer.render(scene,camera);
                const blob = await new Promise((resolve,reject)=>renderer.domElement.toBlob(b=>b?resolve(b):reject(new Error('Normal diagnostic PNG failed')),'image/png'));
                controls.push({kind:flat?'normals-flat':'normals-smooth',direction:'front',blob,anchor:{x:.5,y:1}});
              } finally {scene.remove(object);material.dispose();}
            }
            return controls;
          } finally {geometry.dispose();renderer.dispose();renderer.forceContextLoss();}
        }
      `,
      resolveDir: process.cwd(),
    },
    bundle: true,
    write: false,
    format: 'iife',
    globalName: 'QualityRenderer',
    platform: 'browser',
  });
  const browser = await chromium.launch({ channel: 'chrome', headless: true });
  const page = await browser.newPage();
  const errors: string[] = [];
  const requests: string[] = [];
  page.on('pageerror', (error) => errors.push(error.message));
  page.on('request', (request) => {
    if (/\.onnx(?:\?|$)/.test(request.url()) || request.method() !== 'GET') requests.push(request.url());
  });
  const reports = [];
  try {
    await page.setContent('<html><body>Saved real mesh quality replay</body></html>');
    await page.addScriptTag({ content: bundle.outputFiles[0].text });
    for (const fixture of fixtures) {
      const mesh = readMesh(fixture.directory);
      const stats = inspectMesh(mesh);
      const hashesBefore = Object.fromEntries(
        Object.entries(mesh).map(([key, value]) => [
          key,
          createHash('sha256').update(new Uint8Array(value.buffer)).digest('hex'),
        ]),
      );
      const outputViews = await page.evaluate(
        async (input) => {
          const decode = (text: string) => Uint8Array.from(atob(text), (c) => c.charCodeAt(0)).buffer;
          const mesh = {
            positions: new Float32Array(decode(input.positions)),
            colors: new Float32Array(decode(input.colors)),
            indices: new Uint32Array(decode(input.indices)),
          };
          const inputSnapshot = {
            positions: mesh.positions.slice(),
            colors: mesh.colors.slice(),
            indices: mesh.indices.slice(),
          };
          const renderer = (
            globalThis as typeof globalThis & {
              QualityRenderer: typeof import('../src/lib/multiview/render') & {
                renderNormalControls: (
                  mesh: import('../src/lib/multiview/geometry').ReconstructedMesh,
                  yaw: number,
                ) => Promise<
                  { kind: string; direction: string; blob: Blob; anchor: { x: number; y: number } }[]
                >;
              };
            }
          ).QualityRenderer;
          const exported: {
            kind: string;
            direction: string;
            anchor: { x: number; y: number };
            dataUrl: string;
          }[] = [];
          for (const kind of ['learned', 'uniform'] as const) {
            const selected =
              kind === 'learned' ? mesh : { ...mesh, colors: new Float32Array(mesh.colors.length).fill(1) };
            const views = await renderer.renderThreeViews(selected, {
              size: 768,
              inputOrientationYawOffset: input.yaw,
            });
            for (const view of views) {
              const dataUrl = await new Promise<string>((resolve, reject) => {
                const reader = new FileReader();
                reader.onerror = () => reject(reader.error);
                reader.onload = () => resolve(String(reader.result));
                reader.readAsDataURL(view.blob);
              });
              exported.push({ kind, direction: view.direction, anchor: view.anchor, dataUrl });
            }
          }
          for (const view of await renderer.renderNormalControls(mesh, input.yaw)) {
            const dataUrl = await new Promise<string>((resolve, reject) => {
              const reader = new FileReader();
              reader.onerror = () => reject(reader.error);
              reader.onload = () => resolve(String(reader.result));
              reader.readAsDataURL(view.blob);
            });
            exported.push({ kind: view.kind, direction: view.direction, anchor: view.anchor, dataUrl });
          }
          for (const key of ['positions', 'colors', 'indices'] as const) {
            if (!mesh[key].every((value, index) => value === inputSnapshot[key][index])) {
              throw new Error(`Rendering mutated the actual browser input mesh: ${key}`);
            }
          }
          return exported;
        },
        {
          positions: Buffer.from(mesh.positions.buffer).toString('base64'),
          colors: Buffer.from(mesh.colors.buffer).toString('base64'),
          indices: Buffer.from(mesh.indices.buffer).toString('base64'),
          yaw: fixture.yaw,
        },
      );
      const images: {
        kind: string;
        direction: string;
        hash: string;
        alphaHash: string;
        opaque: number;
        coloredOpaque: number;
        file: string;
      }[] = [];
      for (const view of outputViews) {
        const bytes = Buffer.from(view.dataUrl.split(',')[1], 'base64');
        const file = path.join(output, `${fixture.name}-${view.kind}-${view.direction}.png`);
        writeFileSync(file, bytes);
        const { data, info: imageInfo } = await sharp(bytes)
          .ensureAlpha()
          .raw()
          .toBuffer({ resolveWithObject: true });
        const alpha = Buffer.alloc(imageInfo.width * imageInfo.height);
        let opaque = 0,
          coloredOpaque = 0;
        for (let i = 0; i < data.length; i += 4) {
          alpha[i / 4] = data[i + 3];
          if (data[i + 3] === 255) {
            opaque++;
            if (data[i] < 254 || data[i + 1] < 254 || data[i + 2] < 254) coloredOpaque++;
          }
        }
        expect(opaque).toBeGreaterThan(500);
        if (view.kind === 'uniform')
          expect(
            coloredOpaque,
            `${fixture.name} ${view.direction} uniform white must not gain blue/black marks`,
          ).toBe(0);
        images.push({
          kind: view.kind,
          direction: view.direction,
          hash: createHash('sha256').update(bytes).digest('hex'),
          alphaHash: createHash('sha256').update(alpha).digest('hex'),
          opaque,
          coloredOpaque,
          file,
        });
      }
      for (const direction of ['left', 'front', 'right']) {
        const learned = images.find((image) => image.direction === direction && image.kind === 'learned')!;
        const uniform = images.find((image) => image.direction === direction && image.kind === 'uniform')!;
        expect(uniform.alphaHash, 'color controls must preserve the same silhouette').toBe(learned.alphaHash);
      }
      const hashesAfter = Object.fromEntries(
        Object.entries(mesh).map(([key, value]) => [
          key,
          createHash('sha256').update(new Uint8Array(value.buffer)).digest('hex'),
        ]),
      );
      expect(hashesAfter).toEqual(hashesBefore);
      expect(stats.nonFinite + stats.invalidIndices + stats.outOfRangeColors).toBe(0);
      reports.push({ ...fixture, stats, images, meshInputHashes: hashesBefore });
      writeFileSync(
        path.join(output, 'measurements.json'),
        JSON.stringify(
          {
            browserVersion: browser.version(),
            test: 'saved real mesh replay, no new AI inference',
            reports,
            errors,
            requests,
          },
          null,
          2,
        ),
      );
    }
    expect(errors).toEqual([]);
    expect(requests).toEqual([]);
    await info.attach('quality measurements', {
      path: path.join(output, 'measurements.json'),
      contentType: 'application/json',
    });
  } finally {
    await browser.close();
  }
});
