// Controlled responses below cover UI/state only; they do not count as real model validation.
test.use({ channel: 'chrome' });
type UIControls = { releaseBackground?: () => void; releaseMulti?: () => void; failAssetWrite?: boolean };
async function installUIResponses(page: import('@playwright/test').Page) {
  const png = await sharp(
    Buffer.from(
      '<svg xmlns="http://www.w3.org/2000/svg" width="64" height="64"><rect x="16" y="8" width="32" height="48" fill="white"/></svg>',
    ),
  )
    .png()
    .toBuffer();
  await page.addInitScript(
    (bytes) => {
      const controls = window as Window & UIControls;
      const Original = Worker;
      class ControlledWorker extends EventTarget {
        onmessage: ((event: MessageEvent) => void) | null = null;
        onerror = null;
        onmessageerror = null;
        stopped = false;
        constructor(readonly workerName: string) {
          super();
        }
        postMessage(message: { id: number }) {
          const send = (data: unknown) => {
            if (!this.stopped) this.onmessage?.(new MessageEvent('message', { data }));
          };
          if (this.workerName === 'sjn-background-removal') {
            controls.releaseBackground = () =>
              send({
                type: 'result',
                id: message.id,
                result: {
                  blob: new Blob([new Uint8Array(bytes)], { type: 'image/png' }),
                  width: 64,
                  height: 64,
                  analysisWidth: 512,
                  analysisHeight: 512,
                  backend: 'wasm',
                  precision: 'fp32',
                  downloadMs: 0,
                  initializationMs: 1,
                  processingMs: 1,
                  inferenceMs: 1,
                  cacheSource: 'memory',
                },
              });
          } else {
            send({
              type: 'progress',
              id: message.id,
              progress: { stage: 'reconstructing', message: '모의 응답 대기' },
            });
            controls.releaseMulti = () =>
              send({
                type: 'mesh',
                id: message.id,
                mesh: {
                  positions: new Float32Array([0.8, 0, 0, -0.4, -0.4, 0, -0.4, 0.4, 0, -0.3, 0, 1]),
                  indices: new Uint32Array([0, 1, 2, 0, 3, 1, 1, 3, 2, 2, 3, 0]),
                  colors: new Float32Array([1, 1, 1, 0.8, 0.8, 0.8, 0.9, 0.9, 0.9, 1, 1, 1]),
                },
                orientation: {
                  yawDegrees: -12,
                  confidence: 0.6,
                  ambiguous: false,
                  symmetryError: 0.02,
                  runnerUpGap: 0.4,
                  sourceFacingMargin: 0.978,
                },
                timings: {
                  downloadMs: 0,
                  initializationMs: 1,
                  processingMs: 1,
                  cacheSource: 'cache',
                  modelId: 'ui-controlled',
                  modelRevision: 'ui-controlled',
                },
              });
          }
        }
        terminate() {
          this.stopped = true;
        }
      }
      Object.defineProperty(globalThis, 'Worker', {
        configurable: true,
        value: function (url: string | URL, options?: WorkerOptions) {
          return options?.name === 'sjn-multiview' || options?.name === 'sjn-background-removal'
            ? new ControlledWorker(options.name)
            : new Original(url, options);
        },
      });
      const add = IDBObjectStore.prototype.add;
      IDBObjectStore.prototype.add = function (value, key) {
        if (controls.failAssetWrite && this.name === 'assets')
          throw new DOMException('검증용 저장 공간 부족', 'QuotaExceededError');
        return key === undefined ? add.call(this, value) : add.call(this, value, key);
      };
    },
    [...png],
  );
}

// Replays an ACTUAL previously inferred mesh through the production client/render/UI.
// It checks camera correction without another model download or inference run.
test('실제 AI 메시 재생: 정면 보정으로 세 PNG 갱신 · 추가 Worker/모델 다운로드 없음', async ({ page }) => {
  const meshDirectory = path.resolve(
    process.env.SJN_MULTIVIEW_REPLAY_DIR ?? 'test-results/multiview-chair-left',
  );
  test.skip(
    !existsSync(path.join(meshDirectory, 'mesh-positions.bin')),
    '실제 메시 fixture가 없으면 먼저 opt-in 실추론 검증을 실행하세요.',
  );
  const encoded = Object.fromEntries(
    ['positions', 'indices', 'colors'].map((field) => [
      field,
      readFileSync(path.join(meshDirectory, `mesh-${field}.bin`)).toString('base64'),
    ]),
  );
  const decodeFloat = (field: string) => {
    const bytes = Buffer.from(encoded[field], 'base64');
    return new Float32Array(bytes.buffer.slice(bytes.byteOffset, bytes.byteOffset + bytes.byteLength));
  };
  const indices = Buffer.from(encoded.indices, 'base64');
  const orientation = estimateFrontOrientation({
    positions: decodeFloat('positions'),
    colors: decodeFloat('colors'),
    indices: new Uint32Array(
      indices.buffer.slice(indices.byteOffset, indices.byteOffset + indices.byteLength),
    ),
  });
  const form = await openUIFixture(page);
  await page.evaluate(
    ({ encoded, orientation }) => {
      const Original = Worker;
      const state = window as Window & { multiviewWorkerCount?: number };
      state.multiviewWorkerCount = 0;
      class ReplayWorker extends EventTarget {
        onmessage: ((event: MessageEvent) => void) | null = null;
        onerror = null;
        onmessageerror = null;
        stopped = false;
        postMessage(message: { id: number }) {
          const decode = (field: string) =>
            Uint8Array.from(atob(encoded[field]), (character) => character.charCodeAt(0)).buffer;
          queueMicrotask(() => {
            if (!this.stopped)
              this.onmessage?.(
                new MessageEvent('message', {
                  data: {
                    type: 'mesh',
                    id: message.id,
                    mesh: {
                      positions: new Float32Array(decode('positions')),
                      indices: new Uint32Array(decode('indices')),
                      colors: new Float32Array(decode('colors')),
                    },
                    orientation,
                    timings: {
                      downloadMs: 0,
                      initializationMs: 0,
                      processingMs: 0,
                      cacheSource: 'cache',
                      backend: 'webgpu',
                      modelId: 'actual-mesh-replay',
                      modelRevision: 'actual-mesh-replay',
                    },
                  },
                }),
              );
          });
        }
        terminate() {
          this.stopped = true;
        }
      }
      Object.defineProperty(globalThis, 'Worker', {
        configurable: true,
        value: function (url: string | URL, options?: WorkerOptions) {
          if (options?.name !== 'sjn-multiview') return new Original(url, options);
          state.multiviewWorkerCount = (state.multiviewWorkerCount ?? 0) + 1;
          return new ReplayWorker();
        },
      });
    },
    { encoded, orientation },
  );
  const models: string[] = [];
  page.on('request', (request) => {
    if (/\.onnx(?:\?|$)/.test(request.url())) models.push(request.url());
  });
  await form.getByRole('button', { name: '왼쪽 측면 사진 AI 3방향 생성', exact: true }).click();
  const dialog = page.getByRole('dialog', { name: 'AI 3방향 이미지 생성', exact: true });
  await dialog.getByRole('button', { name: '3방향 생성 시작', exact: true }).click();
  await expect(dialog.getByTestId('multiview-result-front')).toBeVisible();
  const hashes = async () =>
    Promise.all(
      ['left', 'front', 'right'].map(async (direction) => {
        return dialog
          .getByTestId(`multiview-result-${direction}`)
          .evaluate(async (image: HTMLImageElement) => {
            const blob = await (await fetch(image.src)).blob();
            const digest = await crypto.subtle.digest('SHA-256', await blob.arrayBuffer());
            const bitmap = await createImageBitmap(blob);
            const canvas = new OffscreenCanvas(bitmap.width, bitmap.height);
            const ctx = canvas.getContext('2d')!;
            ctx.drawImage(bitmap, 0, 0);
            bitmap.close();
            const pixels = ctx.getImageData(0, 0, canvas.width, canvas.height).data;
            let transparent = 0,
              opaque = 0;
            for (let i = 3; i < pixels.length; i += 4) {
              if (pixels[i] === 0) transparent++;
              if (pixels[i] > 200) opaque++;
            }
            return {
              hash: [...new Uint8Array(digest)].map((n) => n.toString(16).padStart(2, '0')).join(''),
              transparent,
              opaque,
            };
          });
      }),
    );
  await expect(dialog.getByTestId('front-orientation')).toHaveAttribute(
    'data-yaw-degrees',
    String(orientation.yawDegrees),
  );
  const before = await hashes();
  const calibrationOutput = path.resolve('test-results/multiview-calibration');
  mkdirSync(calibrationOutput, { recursive: true });
  await dialog
    .getByRole('region', { name: '정면 생성 결과 확대 보기', exact: true })
    .screenshot({ path: path.join(calibrationOutput, 'front-0.png') });
  const control = dialog.getByLabel('정면 각도 추가 보정', { exact: true });
  await control.focus();
  for (let i = 0; i < 30; i++) {
    await control.press('ArrowRight');
    await expect(control).toBeEnabled();
    if (i === 14 || i === 29)
      await dialog
        .getByRole('region', { name: '정면 생성 결과 확대 보기', exact: true })
        .screenshot({ path: path.join(calibrationOutput, `front-${i + 1}.png`) });
  }
  await expect(dialog.getByTestId('multiview-yaw-value')).toHaveText('+30°');
  await expect(dialog.getByRole('button', { name: '3방향 모두 적용', exact: true })).toBeEnabled();
  const after = await hashes();
  for (let i = 0; i < 3; i++) {
    expect(after[i].hash).not.toBe(before[i].hash);
    expect(after[i].transparent).toBeGreaterThan(50_000);
    expect(after[i].opaque).toBeGreaterThan(5_000);
  }
  expect(
    await page.evaluate(() => (window as Window & { multiviewWorkerCount?: number }).multiviewWorkerCount),
  ).toBe(1);
  expect(models).toEqual([]);
  writeFileSync(
    path.join(calibrationOutput, 'measurements.json'),
    JSON.stringify({ before, after, correction: 30, workers: 1, modelRequests: models }, null, 2),
  );
  for (let i = 0; i < 15; i++) {
    await control.press('ArrowRight');
    await expect(control).toBeEnabled();
  }
  await dialog
    .getByRole('region', { name: '정면 생성 결과 확대 보기', exact: true })
    .screenshot({ path: path.join(calibrationOutput, 'front-45.png') });
  for (let i = 0; i < 15; i++) {
    await control.press('ArrowLeft');
    await expect(control).toBeEnabled();
  }
  await expect(dialog.getByTestId('multiview-yaw-value')).toHaveText('+30°');
  await dialog.getByRole('button', { name: '3방향 모두 적용', exact: true }).click();
  await expect(dialog).toHaveCount(0);
  await expect(form.getByRole('img', { name: '배치 기준점을 지정할 제품 이미지', exact: true })).toHaveCount(
    3,
  );
});
