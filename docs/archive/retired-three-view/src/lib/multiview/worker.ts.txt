import type * as Ort from 'onnxruntime-web';
import { loadMultiViewModel } from './model-cache';
import {
  MULTIVIEW_MODEL_ID,
  MULTIVIEW_PART_LABELS,
  MULTIVIEW_MODEL_REVISION,
  MULTIVIEW_RUNTIME_URL,
  type MultiViewModelPart,
} from './model';
import { foregroundBounds, rgbNchw, transposeTokens } from './pixels';
import { extractMesh, refineMeshSurface, sampleSurfaceColors } from './geometry';
import { estimateFrontOrientation } from './orientation';
import type { MultiViewProgress, MultiViewReply, MultiViewRequest, MultiViewTimings } from './types';

const scope = globalThis as unknown as {
  onmessage: ((event: MessageEvent<MultiViewRequest>) => void) | null;
  postMessage: (reply: MultiViewReply, transfer?: Transferable[]) => void;
};
let running = false;
let fallbackAllowed = false;
let attemptTimings: MultiViewTimings | undefined;
let attemptStarted = 0;
const GRID = 160;
const BOUND = 0.87;
const CHUNK = 8192;

async function prepare(blob: Blob) {
  const bitmap = await createImageBitmap(blob);
  try {
    if (bitmap.width * bitmap.height > 40_000_000)
      throw new Error('제품 사진은 4,000만 화소 이하로 줄여 주세요.');
    const factor = Math.min(1, 2048 / Math.max(bitmap.width, bitmap.height));
    const scan = new OffscreenCanvas(
      Math.max(1, Math.round(bitmap.width * factor)),
      Math.max(1, Math.round(bitmap.height * factor)),
    );
    const ctx = scan.getContext('2d', { willReadFrequently: true });
    if (!ctx) throw new Error('이 브라우저에서 이미지 준비 기능을 사용할 수 없어요.');
    ctx.drawImage(bitmap, 0, 0, scan.width, scan.height);
    const bounds = foregroundBounds(
      ctx.getImageData(0, 0, scan.width, scan.height).data,
      scan.width,
      scan.height,
    );
    const input = new OffscreenCanvas(512, 512);
    const out = input.getContext('2d', { willReadFrequently: true })!;
    out.fillStyle = '#808080';
    out.fillRect(0, 0, 512, 512);
    const extent = Math.max(bounds.width, bounds.height) / 0.85;
    const width = (bounds.width / extent) * 512;
    const height = (bounds.height / extent) * 512;
    out.imageSmoothingEnabled = true;
    out.imageSmoothingQuality = 'high';
    out.drawImage(
      bitmap,
      bounds.x / factor,
      bounds.y / factor,
      bounds.width / factor,
      bounds.height / factor,
      (512 - width) / 2,
      (512 - height) / 2,
      width,
      height,
    );
    return rgbNchw(out.getImageData(0, 0, 512, 512).data, 512, 512);
  } finally {
    bitmap.close();
  }
}
async function requireGpu() {
  const gpu = (
    navigator as unknown as {
      gpu?: {
        requestAdapter: () => Promise<{
          features: { has: (key: string) => boolean };
          limits: { maxStorageBufferBindingSize: number };
        } | null>;
      };
    }
  ).gpu;
  if (!gpu)
    throw new Error(
      '3방향 생성은 WebGPU를 지원하는 Chrome 또는 Edge와 GPU가 필요해요. 이 환경에서는 실행할 수 없어요.',
    );
  const adapter = await gpu.requestAdapter();
  if (!adapter || !adapter.features.has('shader-f16'))
    throw new Error(
      '이 GPU/브라우저는 3방향 생성에 필요한 WebGPU FP16 연산을 지원하지 않아요. 다른 기기나 최신 Chrome/Edge에서 실행해 주세요.',
    );
  if (adapter.limits.maxStorageBufferBindingSize < 128 * 1024 * 1024)
    throw new Error('GPU의 버퍼 한도가 이 3방향 모델을 실행하기에 부족해요.');
}
function explain(error: unknown) {
  const text = error instanceof Error ? error.message : String(error);
  if (/out of memory|memory access|allocation|allocat|out of bounds|device\]?\s+(?:is\s+)?lost/i.test(text))
    return '메모리 부족 또는 실행 환경 문제로 3방향 생성을 끝내지 못했어요. 다른 무거운 작업을 줄이고 다시 시도하거나 메모리가 더 큰 기기에서 실행해 주세요.';
  if (/fetch|network|failed to load|importing a module/i.test(text))
    return '모델 실행 파일을 받지 못했어요. 네트워크 연결과 huggingface.co · cdn.jsdelivr.net 접속을 확인해 주세요.';
  return text.slice(0, 700);
}

async function generate(blob: Blob, progress: (value: MultiViewProgress) => void, forceCpu = false) {
  const timings: MultiViewTimings = {
    downloadMs: 0,
    initializationMs: 0,
    processingMs: 0,
    cacheSource: 'cache',
    backend: forceCpu ? 'wasm' : 'webgpu',
    modelId: MULTIVIEW_MODEL_ID,
    modelRevision: MULTIVIEW_MODEL_REVISION,
  };
  attemptTimings = timings;
  progress({ stage: 'checking', message: '투명 배경과 실행 환경을 확인하고 있어요.' });
  const preparationStarted = performance.now();
  const input = await prepare(blob);
  fallbackAllowed = true;
  if (!forceCpu) await requireGpu();
  fallbackAllowed = false;
  timings.processingMs += performance.now() - preparationStarted;
  progress({ stage: 'loading-runtime', message: '브라우저 3방향 AI 실행 모듈을 불러오고 있어요.' });
  const runtimeStarted = performance.now();
  const ort = await import('onnxruntime-web/webgpu');
  ort.env.wasm.wasmPaths = MULTIVIEW_RUNTIME_URL;
  ort.env.wasm.numThreads = 1;
  ort.env.wasm.proxy = false;
  timings.initializationMs += performance.now() - runtimeStarted;
  const create = async (part: MultiViewModelPart) => {
    fallbackAllowed = false;
    const loaded = await loadMultiViewModel(part, progress);
    fallbackAllowed = true;
    timings.downloadMs += loaded.downloadMs;
    timings.initializationMs += loaded.cacheMs;
    if (loaded.cacheSource === 'network') timings.cacheSource = 'network';
    if (loaded.cacheNotice) timings.cacheNotice = loaded.cacheNotice;
    progress({
      stage: 'initializing',
      message: `${MULTIVIEW_PART_LABELS[part]} 모델을 ${forceCpu ? 'CPU' : 'GPU'}에 준비하고 있어요.`,
    });
    const started = performance.now();
    try {
      return await ort.InferenceSession.create(loaded.bytes, {
        executionProviders: forceCpu
          ? ['wasm']
          : [{ name: 'webgpu', storageBufferCacheMode: 'simple', defaultBufferCacheMode: 'simple' }],
        enableGraphCapture: false,
        graphOptimizationLevel: 'all',
        executionMode: 'sequential',
      });
    } finally {
      timings.initializationMs += performance.now() - started;
    }
  };
  const read = async (tensor: Ort.Tensor) => {
    const values = await tensor.getData();
    if (!(values instanceof Float32Array)) throw new Error('AI 모델의 출력 숫자 형식이 맞지 않아요.');
    return values;
  };
  let tokens: Float32Array;
  const encoder = await create('encoder');
  try {
    progress({ stage: 'encoding', message: '제품 사진의 형태와 색상을 분석하고 있어요.' });
    const started = performance.now();
    const tensor = new ort.Tensor('float32', input, [1, 3, 512, 512]);
    let outputs: Ort.InferenceSession.ReturnType | undefined;
    try {
      outputs = await encoder.run({ image: tensor });
      if (!outputs.tokens) throw new Error('이미지 분석 모델의 출력이 맞지 않아요.');
      tokens = transposeTokens(await read(outputs.tokens));
    } finally {
      tensor.dispose();
      Object.values(outputs ?? {}).forEach((t) => t.dispose());
    }
    timings.processingMs += performance.now() - started;
  } finally {
    await encoder.release();
  }
  let scene: Float32Array;
  const backbone = await create('backbone');
  try {
    progress({ stage: 'reconstructing', message: '같은 제품의 입체 형태를 복원하고 있어요.' });
    const started = performance.now();
    const tensor = new ort.Tensor('float32', tokens!, [1, 1025, 768]);
    let outputs: Ort.InferenceSession.ReturnType | undefined;
    try {
      outputs = await backbone.run({ image_tokens: tensor });
      if (!outputs.triplane) throw new Error('제품 복원 모델의 출력이 맞지 않아요.');
      scene = new Float32Array(await read(outputs.triplane));
      if (scene.length !== 3 * 40 * 64 * 64) throw new Error('복원된 제품 데이터 크기가 맞지 않아요.');
    } finally {
      tensor.dispose();
      Object.values(outputs ?? {}).forEach((t) => t.dispose());
    }
    timings.processingMs += performance.now() - started;
  } finally {
    await backbone.release();
  }
  const decoder = await create('decoder');
  const triplane = new ort.Tensor('float32', scene!, [3, 40, 64, 64]);
  try {
    const started = performance.now();
    const query = async (points: Float32Array, field: 'density' | 'color') => {
      const tensor = new ort.Tensor('float32', points, [CHUNK, 3]);
      let outputs: Ort.InferenceSession.ReturnType | undefined;
      try {
        outputs = await decoder.run({ triplane, points: tensor }, [field]);
        if (!outputs[field]) throw new Error('제품 표면 모델의 출력이 맞지 않아요.');
        const values = new Float32Array(await read(outputs[field]));
        if (values.length !== CHUNK * (field === 'color' ? 3 : 1))
          throw new Error('제품 표면 모델의 출력 크기가 맞지 않아요.');
        return values;
      } finally {
        tensor.dispose();
        Object.values(outputs ?? {}).forEach((t) => t.dispose());
      }
    };
    const density = new Float32Array(GRID ** 3);
    for (let offset = 0; offset < density.length; offset += CHUNK) {
      const points = new Float32Array(CHUNK * 3);
      const count = Math.min(CHUNK, density.length - offset);
      for (let i = 0; i < count; i++) {
        const id = offset + i;
        points[i * 3] = (Math.floor(id / (GRID * GRID)) / (GRID - 1)) * 2 * BOUND - BOUND;
        points[i * 3 + 1] = ((Math.floor(id / GRID) % GRID) / (GRID - 1)) * 2 * BOUND - BOUND;
        points[i * 3 + 2] = ((id % GRID) / (GRID - 1)) * 2 * BOUND - BOUND;
      }
      const values = await query(points, 'density');
      for (let i = 0; i < count; i++) {
        if (Number.isNaN(values[i]) || values[i] < 0)
          throw new Error(
            '제품 표면 복원에서 유효하지 않은 숫자가 나왔어요. 다른 사진으로 다시 시도해 주세요.',
          );
        // FP16 exp can overflow inside solid areas; keep interpolation finite.
        density[offset + i] = Math.max(0, Math.min(1e6, values[i]));
      }
      progress({
        stage: 'geometry',
        message: '제품의 곡면과 가장자리를 복원하고 있어요.',
        completed: offset + count,
        total: density.length,
      });
    }
    // The ONNX decoder already exports exp(raw - 1), not raw density.
    const mesh = extractMesh(density, undefined, GRID, BOUND, 25);
    if (mesh.positions.length < 9)
      throw new Error(
        '사진에서 제품의 입체 형태를 복원하지 못했어요. 제품 전체가 선명하게 보이는 사진을 사용해 주세요.',
      );
    const positions = await refineMeshSurface(
      mesh,
      density,
      GRID,
      BOUND,
      25,
      (points) => query(points, 'density'),
      {
        iterations: 7,
        chunkSize: CHUNK,
        onProgress: (completed, total) =>
          progress({
            stage: 'geometry',
            message: '제품 표면을 정밀하게 맞추고 있어요.',
            completed,
            total,
          }),
      },
    );
    const colors = await sampleSurfaceColors(positions, (points) => query(points, 'color'), {
      chunkSize: CHUNK,
      onProgress: (completed, total) =>
        progress({
          stage: 'coloring',
          message: '복원한 제품에 동일한 색상과 무늬를 입히고 있어요.',
          completed,
          total,
        }),
    });
    timings.processingMs += performance.now() - started;
    return { mesh: { positions, indices: mesh.indices, colors }, timings };
  } finally {
    triplane.dispose();
    await decoder.release();
  }
}
scope.onmessage = async (event) => {
  const message = event.data;
  if (message.type !== 'run') return;
  if (running) {
    scope.postMessage({ type: 'error', id: message.id, message: '이미 3방향 생성이 진행 중이에요.' });
    return;
  }
  running = true;
  fallbackAllowed = false;
  attemptTimings = undefined;
  attemptStarted = performance.now();
  try {
    const { mesh, timings } = await generate(
      message.blob,
      (progress) => scope.postMessage({ type: 'progress', id: message.id, progress }),
      message.forceCpu,
    );
    // Inference sessions are released before this CPU-only analysis. A geometry
    // analysis error cannot be recovered by downloading/re-running on CPU.
    fallbackAllowed = false;
    scope.postMessage({
      type: 'progress',
      id: message.id,
      progress: {
        stage: 'aligning',
        message: '제품의 좌우 형태를 비교해 정면 각도를 찾고 있어요.',
      },
    });
    const alignmentStarted = performance.now();
    const orientation = estimateFrontOrientation(mesh);
    timings.processingMs += performance.now() - alignmentStarted;
    scope.postMessage({ type: 'mesh', id: message.id, mesh, orientation, timings }, [
      mesh.positions.buffer,
      mesh.indices.buffer,
      mesh.colors.buffer,
    ] as ArrayBuffer[]);
  } catch (error) {
    const failedTimings = attemptTimings as MultiViewTimings | undefined;
    if (failedTimings)
      failedTimings.processingMs = Math.max(
        failedTimings.processingMs,
        performance.now() - attemptStarted - failedTimings.downloadMs - failedTimings.initializationMs,
      );
    scope.postMessage({
      type: 'error',
      id: message.id,
      message: explain(error),
      retryCpu: !message.forceCpu && fallbackAllowed,
      timings: attemptTimings,
    });
  } finally {
    running = false;
  }
};
