import type {
  MultiViewMesh,
  MultiViewOptions,
  MultiViewProgress,
  MultiViewReply,
  MultiViewResult,
  MultiViewTimings,
} from './types';
import { renderThreeViews } from './render';
import type { FrontOrientationEstimate } from './orientation';

interface Pending {
  id: number;
  blob?: Blob;
  cpuRetried: boolean;
  carryTimings?: MultiViewTimings;
  options: MultiViewOptions;
  onProgress: (value: MultiViewProgress) => void;
  resolve: (value: MultiViewResult) => void;
  reject: (error: Error) => void;
  abort: AbortController;
}
function validate(options: MultiViewOptions) {
  const correction = options.yawAdjustmentDegrees ?? 0;
  if (!Number.isFinite(correction) || Math.abs(correction) > 180)
    throw new Error('정면 보정 각도는 -180도부터 180도 사이여야 해요.');
}
export class MultiViewClient {
  private worker?: Worker;
  private pending?: Pending;
  private sequence = 0;
  // Keep only the small reconstructed mesh, never model weights or GPU sessions.
  private reconstructed?: {
    mesh: MultiViewMesh;
    orientation: FrontOrientationEstimate;
    timings: MultiViewTimings;
  };

  run(blob: Blob, options: MultiViewOptions, onProgress: Pending['onProgress']): Promise<MultiViewResult> {
    if (this.pending)
      return Promise.reject(new Error('3방향 생성이 진행 중이에요. 완료하거나 취소해 주세요.'));
    if (typeof Worker === 'undefined')
      return Promise.reject(new Error('이 브라우저는 Web Worker를 지원하지 않아요.'));
    if (!blob.size || blob.size > 100 * 1024 * 1024)
      return Promise.reject(new Error('제품 사진이 비어 있거나 처리 가능한 크기를 초과했어요.'));
    try {
      validate(options);
    } catch (error) {
      return Promise.reject(error);
    }
    this.reconstructed = undefined;
    return new Promise((resolve, reject) => {
      this.pending = {
        id: ++this.sequence,
        blob,
        cpuRetried: false,
        options: { ...options },
        onProgress,
        resolve,
        reject,
        abort: new AbortController(),
      };
      this.startWorker();
    });
  }
  /** Reframe the same inferred product without downloads or another AI inference. */
  rerender(options: MultiViewOptions, onProgress: Pending['onProgress']): Promise<MultiViewResult> {
    if (this.pending) return Promise.reject(new Error('이미지를 갱신하고 있어요. 잠시 기다려 주세요.'));
    if (!this.reconstructed) return Promise.reject(new Error('먼저 제품의 세 방향을 생성해 주세요.'));
    try {
      validate(options);
    } catch (error) {
      return Promise.reject(error);
    }
    return new Promise((resolve, reject) => {
      const pending: Pending = {
        id: ++this.sequence,
        cpuRetried: false,
        options: { ...options },
        onProgress,
        resolve,
        reject,
        abort: new AbortController(),
      };
      this.pending = pending;
      this.render(pending);
    });
  }
  private startWorker(forceCpu = false) {
    const job = this.pending;
    if (!job?.blob) return;
    try {
      const worker = new Worker(new URL('./worker.ts', import.meta.url), {
        type: 'module',
        name: 'sjn-multiview',
      });
      this.worker = worker;
      worker.onerror = (event) => {
        event.preventDefault();
        if (this.worker === worker)
          this.fail('3방향 AI 작업이 중단됐어요. 메모리 또는 브라우저 지원을 확인해 주세요.');
      };
      worker.onmessageerror = () => {
        if (this.worker === worker) this.fail('AI 작업 결과를 읽지 못했어요. 다시 시도해 주세요.');
      };
      worker.onmessage = (event: MessageEvent<MultiViewReply>) => {
        const pending = this.pending;
        if (!pending || this.worker !== worker) return;
        const message = event.data;
        if (!message || typeof message !== 'object' || !Number.isInteger(message.id)) {
          this.fail('AI 작업 결과 형식이 올바르지 않아요. 다시 생성해 주세요.');
          return;
        }
        if (pending.id !== message.id) return;
        if (message.type === 'progress') {
          if (!message.progress || typeof message.progress.message !== 'string') {
            this.fail('AI 진행 상태를 읽지 못했어요. 다시 생성해 주세요.');
            return;
          }
          pending.onProgress(message.progress);
          return;
        }
        if (message.type === 'error') {
          if (typeof message.message !== 'string') {
            this.fail('AI 작업 오류를 읽지 못했어요. 다시 생성해 주세요.');
            return;
          }
          if (message.retryCpu && !pending.cpuRetried) {
            pending.cpuRetried = true;
            pending.carryTimings = message.timings;
            worker.terminate();
            this.worker = undefined;
            pending.onProgress({
              stage: 'initializing',
              message: 'GPU 처리를 완료하지 못해 같은 모델을 CPU로 실행해요. 수 분 이상 걸릴 수 있어요.',
            });
            if (this.pending === pending) this.startWorker(true);
            return;
          }
          this.fail(message.message);
          return;
        }
        worker.terminate();
        this.worker = undefined;
        if (!message.orientation || !Number.isFinite(message.orientation.yawDegrees)) {
          this.fail('제품의 정면 분석 결과를 읽지 못했어요. 페이지를 새로고침한 뒤 다시 생성해 주세요.');
          return;
        }
        if (
          message.type !== 'mesh' ||
          !(message.mesh?.positions instanceof Float32Array) ||
          !(message.mesh?.indices instanceof Uint32Array) ||
          !(message.mesh?.colors instanceof Float32Array) ||
          !message.timings ||
          ![message.timings.downloadMs, message.timings.initializationMs, message.timings.processingMs].every(
            (value) => Number.isFinite(value) && value >= 0,
          ) ||
          !['network', 'cache'].includes(message.timings.cacheSource) ||
          typeof message.timings.modelId !== 'string' ||
          typeof message.timings.modelRevision !== 'string' ||
          ![
            message.orientation.confidence,
            message.orientation.sourceFacingMargin,
            message.orientation.runnerUpGap,
          ].every((value) => Number.isFinite(value) && value >= 0 && value <= 1) ||
          !Number.isFinite(message.orientation.symmetryError) ||
          message.orientation.symmetryError < 0 ||
          typeof message.orientation.ambiguous !== 'boolean'
        ) {
          this.fail('AI 작업 결과가 불완전해요. 다시 생성해 주세요.');
          return;
        }
        const carry = pending.carryTimings;
        this.reconstructed = {
          mesh: message.mesh,
          orientation: { ...message.orientation },
          timings: {
            ...message.timings,
            downloadMs: message.timings.downloadMs + (carry?.downloadMs ?? 0),
            initializationMs: message.timings.initializationMs + (carry?.initializationMs ?? 0),
            processingMs: message.timings.processingMs + (carry?.processingMs ?? 0),
            cacheSource: carry?.cacheSource === 'network' ? 'network' : message.timings.cacheSource,
            cacheNotice: message.timings.cacheNotice ?? carry?.cacheNotice,
          },
        };
        this.render(pending);
      };
      worker.postMessage({
        type: 'run',
        id: job.id,
        blob: job.blob,
        ...(forceCpu ? { forceCpu: true } : {}),
      });
    } catch (error) {
      this.fail(error instanceof Error ? error.message : 'AI 작업을 시작하지 못했어요.');
    }
  }
  private render(pending: Pending) {
    const reconstructed = this.reconstructed!;
    pending.onProgress({
      stage: 'rendering',
      message: '추정한 정면을 먼저 만들고, 같은 기준으로 좌우 이미지를 만들고 있어요.',
    });
    const started = performance.now();
    void renderThreeViews(reconstructed.mesh, {
      inputOrientationYawOffset:
        reconstructed.orientation.yawDegrees + (pending.options.yawAdjustmentDegrees ?? 0),
      size: 768,
      elevationDegrees: 10,
      signal: pending.abort.signal,
      onProgress: (completed, total) => {
        if (this.pending !== pending) return;
        pending.onProgress({
          stage: 'rendering',
          message:
            completed === 1
              ? '정면 이미지를 만들었어요. 같은 기준으로 좌우를 만들고 있어요.'
              : '방향별 투명 PNG를 준비하고 있어요.',
          completed,
          total,
        });
      },
    })
      .then((views) => {
        if (this.pending !== pending) return;
        this.pending = undefined;
        pending.resolve({
          ...reconstructed.timings,
          orientation: { ...reconstructed.orientation },
          processingMs: reconstructed.timings.processingMs + performance.now() - started,
          views,
        });
      })
      .catch((error: unknown) => {
        if (this.pending === pending)
          this.fail(error instanceof Error ? error.message : '3방향 이미지 렌더링에 실패했어요.');
      });
  }
  private fail(message: string) {
    const pending = this.pending;
    this.pending = undefined;
    pending?.abort.abort();
    this.worker?.terminate();
    this.worker = undefined;
    pending?.reject(new Error(message));
  }
  dispose() {
    this.fail('3방향 생성을 취소했어요.');
    this.reconstructed = undefined;
  }
}
