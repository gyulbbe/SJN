'use client';

import { useEffect, useRef, useState } from 'react';
import type { MultiViewClient } from '@/lib/multiview/client';
import type { MultiViewProgress, MultiViewResult } from '@/lib/multiview/types';
import { resolveBackgroundRemovalInput } from '@/lib/background-removal/source';
import styles from './multiview-generator.module.css';

type Direction = 'left' | 'front' | 'right';
type Source = { blob: Blob; url: string; name: string };
const DIRECTIONS = [
  { value: 'front', label: '정면' },
  { value: 'left', label: '왼쪽' },
  { value: 'right', label: '오른쪽' },
] as const;
const seconds = (value: number) => `${(value / 1000).toFixed(2)}초`;
const signedDegrees = (value: number) => `${value > 0 ? '+' : ''}${Number(value.toFixed(1))}°`;

export function MultiViewGenerator({
  assetId,
  blob,
  onApply,
  canApply = true,
  onClose,
}: {
  assetId: string;
  direction: string;
  /** An unsaved background-removal result can be used without running that model again. */
  blob?: Blob;
  onApply: (result: MultiViewResult) => Promise<void>;
  canApply?: boolean;
  onClose: () => void;
}) {
  const dialogRef = useRef<HTMLDialogElement>(null);
  const clientRef = useRef<MultiViewClient | null>(null);
  const runRef = useRef(0);
  const applyingRef = useRef(false);
  const busyRef = useRef(false);
  const aliveRef = useRef(true);
  const resultUrlsRef = useRef<string[]>([]);
  const committedYawRef = useRef(0);
  const [source, setSource] = useState<Source>();
  const [sourceAttempt, setSourceAttempt] = useState(0);
  const [sourceLoading, setSourceLoading] = useState(true);
  const [sourceError, setSourceError] = useState('');
  const [busy, setBusy] = useState(false);
  const [applying, setApplying] = useState(false);
  const [error, setError] = useState('');
  const [applyError, setApplyError] = useState('');
  const [progress, setProgress] = useState<MultiViewProgress>();
  const [result, setResult] = useState<MultiViewResult>();
  const [resultUrls, setResultUrls] = useState<Partial<Record<Direction, string>>>({});
  const [background, setBackground] = useState<'checker' | 'white' | 'black'>('checker');
  const [zoom, setZoom] = useState(100);
  const [yawAdjustment, setYawAdjustment] = useState(0);
  const [adjusting, setAdjusting] = useState(false);
  const [retryYaw, setRetryYaw] = useState<number>();

  useEffect(() => {
    aliveRef.current = true;
    const dialog = dialogRef.current;
    dialog?.showModal();
    return () => {
      aliveRef.current = false;
      clientRef.current?.dispose();
      clientRef.current = null;
      resultUrlsRef.current.forEach((url) => URL.revokeObjectURL(url));
      resultUrlsRef.current = [];
      dialog?.close();
    };
  }, []);

  useEffect(() => {
    let active = true;
    let url: string | undefined;
    setSourceLoading(true);
    setSourceError('');
    setSource(undefined);
    void (async () => {
      const { asset } = await resolveBackgroundRemovalInput(assetId);
      if (!active) return;
      const input = blob ?? asset.blob;
      url = URL.createObjectURL(input);
      setSource({ blob: input, url, name: asset.name });
    })()
      .catch((reason: unknown) => {
        if (active)
          setSourceError(reason instanceof Error ? reason.message : '제품 사진을 불러오지 못했어요.');
      })
      .finally(() => {
        if (active) setSourceLoading(false);
      });
    return () => {
      active = false;
      if (url) URL.revokeObjectURL(url);
    };
  }, [assetId, blob, sourceAttempt]);

  const clearResults = () => {
    resultUrlsRef.current.forEach((url) => URL.revokeObjectURL(url));
    resultUrlsRef.current = [];
    setResultUrls({});
    setResult(undefined);
    committedYawRef.current = 0;
    setYawAdjustment(0);
    setRetryYaw(undefined);
  };
  const publishResult = (completed: MultiViewResult) => {
    const urls: Partial<Record<Direction, string>> = {};
    const created: string[] = [];
    try {
      for (const view of completed.views) {
        const url = URL.createObjectURL(view.blob);
        created.push(url);
        urls[view.direction] = url;
      }
    } catch (reason) {
      created.forEach((url) => URL.revokeObjectURL(url));
      throw reason;
    }
    resultUrlsRef.current.forEach((url) => URL.revokeObjectURL(url));
    resultUrlsRef.current = created;
    setResultUrls(urls);
    setResult(completed);
  };
  const generate = async () => {
    if (!source || busyRef.current || applyingRef.current) return;
    busyRef.current = true;
    const run = ++runRef.current;
    clientRef.current?.dispose();
    clientRef.current = null;
    setBusy(true);
    setAdjusting(false);
    setError('');
    setApplyError('');
    clearResults();
    setProgress(undefined);
    try {
      const { MultiViewClient: Client } = await import('@/lib/multiview/client');
      if (run !== runRef.current || !aliveRef.current) return;
      const client = new Client();
      clientRef.current = client;
      const completed = await client.run(source.blob, {}, (next) => {
        if (run === runRef.current && aliveRef.current) setProgress(next);
      });
      if (run !== runRef.current || !aliveRef.current) return;
      publishResult(completed);
      setZoom(100);
    } catch (reason) {
      if (run === runRef.current && aliveRef.current)
        setError(reason instanceof Error ? reason.message : '3방향 생성에 실패했어요. 다시 시도해 주세요.');
    } finally {
      if (run === runRef.current && aliveRef.current) {
        busyRef.current = false;
        setBusy(false);
      }
    }
  };
  const adjustYaw = async (nextYaw: number) => {
    if (
      !result ||
      !clientRef.current ||
      busyRef.current ||
      applyingRef.current ||
      nextYaw === committedYawRef.current
    )
      return;
    if (!Number.isFinite(nextYaw) || nextYaw < -180 || nextYaw > 180) return;
    busyRef.current = true;
    const run = ++runRef.current;
    setBusy(true);
    setAdjusting(true);
    setYawAdjustment(nextYaw);
    setError('');
    setRetryYaw(undefined);
    setApplyError('');
    setProgress({ stage: 'rendering', message: '보정한 정면을 기준으로 세 방향을 함께 갱신하고 있어요.' });
    try {
      const completed = await clientRef.current.rerender({ yawAdjustmentDegrees: nextYaw }, (next) => {
        if (run === runRef.current && aliveRef.current) setProgress(next);
      });
      if (run !== runRef.current || !aliveRef.current) return;
      publishResult(completed);
      committedYawRef.current = nextYaw;
    } catch (reason) {
      if (run === runRef.current && aliveRef.current) {
        setError(reason instanceof Error ? reason.message : '정면 각도 보정을 완료하지 못했어요.');
        setYawAdjustment(committedYawRef.current);
        setRetryYaw(nextYaw);
      }
    } finally {
      if (run === runRef.current && aliveRef.current) {
        busyRef.current = false;
        setBusy(false);
        setAdjusting(false);
      }
    }
  };
  const cancel = () => {
    busyRef.current = false;
    runRef.current++;
    clientRef.current?.dispose();
    clientRef.current = null;
    setBusy(false);
    setProgress(undefined);
    setError('생성을 취소했어요. 저장된 모델 캐시는 다음 실행에서 다시 사용해요.');
  };
  const close = () => {
    if (applyingRef.current) return;
    runRef.current++;
    clientRef.current?.dispose();
    onClose();
  };
  const download = (viewDirection: Direction) => {
    const url = resultUrls[viewDirection];
    if (!url || !source || busyRef.current || yawAdjustment !== committedYawRef.current) return;
    const anchor = document.createElement('a');
    anchor.href = url;
    anchor.download = `${source.name.replace(/\.[^/.]+$/, '')}_AI_${DIRECTIONS.find((item) => item.value === viewDirection)!.label}.png`;
    document.body.append(anchor);
    anchor.click();
    anchor.remove();
  };
  const apply = async () => {
    if (
      !result ||
      busyRef.current ||
      !canApply ||
      applyingRef.current ||
      yawAdjustment !== committedYawRef.current
    )
      return;
    applyingRef.current = true;
    setApplying(true);
    setApplyError('');
    try {
      await onApply(result);
      clientRef.current?.dispose();
      onClose();
    } catch (reason) {
      if (aliveRef.current)
        setApplyError(reason instanceof Error ? reason.message : '3방향 이미지를 적용하지 못했어요.');
    } finally {
      applyingRef.current = false;
      if (aliveRef.current) setApplying(false);
    }
  };
  const downloadPercent =
    progress?.totalBytes && progress.loadedBytes !== undefined
      ? Math.min(100, (progress.loadedBytes / progress.totalBytes) * 100)
      : undefined;

  return (
    <dialog
      ref={dialogRef}
      className={styles.dialog}
      aria-labelledby="multiview-title"
      aria-describedby="multiview-description"
      onCancel={(event) => {
        event.preventDefault();
        close();
      }}
    >
      <header className={styles.header}>
        <div>
          <span className={styles.eyebrow}>제품 사진 · 브라우저에서 AI 생성</span>
          <h2 id="multiview-title">AI 3방향 이미지 생성</h2>
          <p id="multiview-description">
            사진에서 제품의 정면을 먼저 추정하고, 그 정면을 기준으로 왼쪽·오른쪽 이미지를 만들어요.
          </p>
        </div>
        <button type="button" className="btn" disabled={applying} onClick={close}>
          {busy ? '취소하고 닫기' : '닫기'}
        </button>
      </header>
      <div className={styles.content}>
        <section className={styles.setup} aria-label="생성할 제품과 자동 정면 정렬">
          <div className={`${styles.source} ${styles.checker}`}>
            {source ? (
              <img src={source.url} alt="3방향 생성에 사용할 제품 사진" data-testid="multiview-source" />
            ) : (
              <span>사진 준비 중…</span>
            )}
          </div>
          <div className={styles.settings}>
            <h3>제품의 정면을 자동으로 맞춰요</h3>
            <p className={styles.note}>
              사진이 조금 비스듬하거나 옆을 보고 있어도, 제품의 입체 형태를 분석해 정면을 먼저 추정해요.
              왼쪽·오른쪽 이미지는 같은 정면을 기준으로 만들어요.
            </p>
            <p className={styles.note}>
              좌우 대칭이 약하거나 앞뒤 모양이 비슷한 제품은 정면이 모호할 수 있어요. 생성한 정면을 확인하고
              필요하면 추가로 각도를 조정할 수 있어요.
            </p>
            <p className={styles.note}>
              {blob
                ? '방금 배경을 제거한 결과를 바로 사용해요.'
                : '배경을 지운 투명 제품 사진이 필요해요. 배경이 남아 있다면 AI 배경 제거부터 진행해 주세요.'}
            </p>
            <div className={styles.requirements}>
              <strong>이 기기에서 GPU를 우선 사용해요</strong>
              <p>
                처음 실행할 때 모델 약 840MB를 내려받으며, 저장된 모델은 다시 사용해요. 기기에 따라 수 분 이상
                걸릴 수 있어요. GPU 실행이 어려우면 같은 모델을 CPU로 처리해요. CPU 처리는 약 4GB 이상의 여유
                메모리와 더 긴 시간이 필요할 수 있어요.
              </p>
              <p>사진은 브라우저에서 처리하며 생성 서버로 전송하지 않아요.</p>
            </div>
            <button
              type="button"
              className="btn primary"
              disabled={!source || sourceLoading || busy || applying}
              onClick={generate}
            >
              {result ? '세 방향 다시 생성' : busy ? '3방향 생성 중…' : '3방향 생성 시작'}
            </button>
          </div>
        </section>
        {sourceError && (
          <div className={styles.error} role="alert">
            <strong>사진을 준비하지 못했어요.</strong>
            <p>{sourceError}</p>
            <button type="button" className="btn" onClick={() => setSourceAttempt((value) => value + 1)}>
              사진 다시 불러오기
            </button>
          </div>
        )}
        {busy && (
          <div className={styles.progress} role="status" aria-live="polite">
            <div>
              <strong>{progress?.message ?? 'AI 실행 모듈을 준비하고 있어요.'}</strong>
              {progress?.loadedBytes !== undefined && (
                <>
                  <progress max="100" value={downloadPercent} aria-label="3방향 모델 다운로드 진행" />
                  <span>
                    {(progress.loadedBytes / 1_048_576).toFixed(1)} MB
                    {progress.totalBytes ? ` / ${(progress.totalBytes / 1_048_576).toFixed(1)} MB` : ''}
                  </span>
                </>
              )}
            </div>
            {!adjusting && (
              <button type="button" className="btn" onClick={cancel}>
                생성 취소
              </button>
            )}
          </div>
        )}
        {error && (
          <div className={styles.error} role="alert">
            <strong>{result ? '기존 결과를 유지하고 있어요.' : '생성 결과가 아직 없어요.'}</strong>
            <p>{error}</p>
            <button
              type="button"
              className="btn"
              disabled={busy || !source}
              onClick={() => (retryYaw === undefined ? generate() : adjustYaw(retryYaw))}
            >
              {retryYaw === undefined ? '다시 시도' : '각도 다시 적용'}
            </button>
          </div>
        )}
        {result && (
          <section className={styles.results} aria-label="생성한 세 방향 결과">
            <div className={styles.toolbar}>
              <h3>정면을 기준으로 만든 세 방향</h3>
              <div className={styles.backgrounds} role="group" aria-label="3방향 결과 배경색">
                {(
                  [
                    ['checker', '체크무늬'],
                    ['white', '흰색'],
                    ['black', '검은색'],
                  ] as const
                ).map(([value, label]) => (
                  <button
                    key={value}
                    type="button"
                    className={`btn small ${background === value ? 'active' : ''}`}
                    aria-pressed={background === value}
                    onClick={() => setBackground(value)}
                  >
                    {label}
                  </button>
                ))}
              </div>
              <label className={styles.zoom}>
                확대
                <select
                  className="input"
                  aria-label="3방향 결과 확대"
                  value={zoom}
                  onChange={(event) => setZoom(Number(event.target.value))}
                >
                  <option value={100}>화면 맞춤</option>
                  <option value={200}>2배</option>
                  <option value={400}>4배</option>
                </select>
              </label>
            </div>
            <div
              className={`${styles.orientation} ${result.orientation.ambiguous ? styles.orientationAmbiguous : ''}`}
              data-testid="front-orientation"
              data-yaw-degrees={result.orientation.yawDegrees}
              data-ambiguous={result.orientation.ambiguous}
            >
              <div className={styles.orientationHeading}>
                <strong>
                  {result.orientation.ambiguous ? '추정한 정면을 확인해 주세요' : '자동 정면 정렬'}
                </strong>
                <span>자동 정렬 각도 {signedDegrees(result.orientation.yawDegrees)}</span>
              </div>
              <p>
                {result.orientation.ambiguous
                  ? '입체 형태에서 정면을 명확히 구분하기 어려워요. 정면 카드가 실제 제품의 앞모습인지 확인하고, 필요하면 아래에서 각도를 조정해 주세요.'
                  : '사진에서 추정한 정면을 먼저 만들고, 같은 기준으로 좌우 이미지를 만들었어요. 세 방향의 제품 형태를 확인해 주세요.'}
              </p>
            </div>
            <div className={styles.angleAdjustment}>
              <div className={styles.angleHeading}>
                <label htmlFor="multiview-yaw">정면 각도 추가 보정</label>
                <output htmlFor="multiview-yaw" data-testid="multiview-yaw-value">
                  {signedDegrees(yawAdjustment)}
                </output>
              </div>
              <input
                id="multiview-yaw"
                type="range"
                min="-180"
                max="180"
                step="1"
                aria-label="정면 각도 추가 보정"
                aria-valuetext={`자동 추정한 정면에서 추가 ${yawAdjustment}도`}
                value={yawAdjustment}
                disabled={busy || applying}
                onChange={(event) => setYawAdjustment(Number(event.target.value))}
                onPointerUp={(event) => {
                  void adjustYaw(Number(event.currentTarget.value));
                }}
                onKeyUp={(event) => {
                  void adjustYaw(Number(event.currentTarget.value));
                }}
                onBlur={(event) => {
                  void adjustYaw(Number(event.currentTarget.value));
                }}
              />
              <p>
                0°는 자동 추정한 정면이에요. 필요할 때만 추가로 돌려 주세요. AI를 다시 실행하지 않고 세 방향을
                함께 갱신해요.
              </p>
            </div>
            <div className={styles.cards}>
              {DIRECTIONS.map(({ value, label }) => {
                const view = result.views.find((item) => item.direction === value);
                return (
                  <article key={value} className={styles.card}>
                    <h4>{label}</h4>
                    <div
                      className={`${styles.viewport} ${styles[background]}`}
                      tabIndex={0}
                      role="region"
                      aria-label={`${label} 생성 결과 확대 보기`}
                    >
                      <div
                        style={{
                          width: `${zoom}%`,
                          minHeight: '100%',
                          display: 'grid',
                          placeItems: 'center',
                        }}
                      >
                        <img
                          src={resultUrls[value]}
                          alt={`AI가 새로 만든 ${label} 제품 사진`}
                          data-testid={`multiview-result-${value}`}
                          draggable={false}
                        />
                      </div>
                    </div>
                    <div className={styles.cardFooter}>
                      <span>
                        {view?.width} × {view?.height} px
                      </span>
                      <button
                        type="button"
                        className="btn small"
                        disabled={busy || applying || yawAdjustment !== committedYawRef.current}
                        onClick={() => download(value)}
                        aria-label={`${label} 투명 PNG 다운로드`}
                      >
                        투명 PNG 다운로드
                      </button>
                    </div>
                  </article>
                );
              })}
            </div>
            <dl className={styles.timings}>
              <div>
                <dt>모델 다운로드</dt>
                <dd data-testid="multiview-download-time">{seconds(result.downloadMs)}</dd>
                <small>{result.cacheSource === 'cache' ? '브라우저 캐시 사용' : '네트워크 사용'}</small>
              </div>
              <div>
                <dt>모델 준비</dt>
                <dd>{seconds(result.initializationMs)}</dd>
              </div>
              <div>
                <dt>세 방향 생성 · {result.backend === 'wasm' ? 'CPU' : 'GPU'}</dt>
                <dd data-testid="multiview-processing-time">{seconds(result.processingMs)}</dd>
              </div>
            </dl>
            {result.cacheNotice && <p className={styles.notice}>{result.cacheNotice}</p>}
            <p className={styles.note}>
              TripoSR가 한 장의 사진에서 추정한 결과예요. 보이지 않던 뒷면·얇은 부품·깊이와 광택은 실제 제품과
              다를 수 있으니 세 방향을 확인한 뒤 적용해 주세요.
            </p>
          </section>
        )}
      </div>
      {applyError && (
        <div className={styles.applyError} role="alert">
          <strong>아직 제품 사진을 교체하지 않았어요.</strong>
          <p>{applyError} 결과는 유지돼요. 다시 적용하거나 PNG를 다운로드할 수 있어요.</p>
        </div>
      )}
      <footer className={styles.footer}>
        <p>
          {applying
            ? '세 방향 이미지를 저장하고 있어요…'
            : '적용하면 기존 왼쪽·정면·오른쪽 사진을 함께 교체해요. 다른 방향은 유지하며, 자재 저장 시 최종 반영돼요.'}
        </p>
        <button
          type="button"
          className="btn primary"
          disabled={!result || busy || applying || !canApply || yawAdjustment !== committedYawRef.current}
          onClick={apply}
        >
          {applying ? '적용 중…' : '3방향 모두 적용'}
        </button>
      </footer>
    </dialog>
  );
}
